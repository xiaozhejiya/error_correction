# 首页搭建 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 搭建 Flutter 首页、基础路由、主题色、资源常量、全局星空背景和工作台占位页。

**Architecture:** 使用轻量分层结构：`main.dart` 只启动应用，`app` 管理应用壳、路由和主题，`core` 放共享常量与通用组件，`features` 放具体页面。首页和工作台都包在同一个 `StarryBackground` 中，保证全局闪烁星星效果一致。

**Tech Stack:** Flutter、Material 3、Widget Test、`flutter_svg`、自定义 `CustomPainter` 和基础 Flutter 动画。

```dart
import 'package:flutter/material.dart';
```

```dart
import '../../features/home/presentation/pages/home_page.dart';
```

```dart
class AppRoutes {
  static const String home = '/';
  static const String workspace = '/workspace';
}

class AppRouter {
  static Route<dynamic> onGenerateRoute(RouteSettings settings) {
    final Widget page = switch (settings.name) {
      AppRoutes.workspace => const WorkspacePage(),
      AppRoutes.home || null => const HomePage(),

      builder: (_) => page,
    );
  }
}
```

```dart
enum GradientActionButtonVariant { primary, secondary }

class GradientActionButton extends StatelessWidget {

    required this.label,
    required this.onPressed,
    this.icon,
    this.variant = GradientActionButtonVariant.primary,

  final VoidCallback onPressed;
  final IconData? icon;

    return Semantics(
      button: true,
      child: Material(
        color: Colors.transparent,
        child: InkWell(
          borderRadius: BorderRadius.circular(14),
          onTap: onPressed,

                      colors: [AppTheme.primary, AppTheme.primaryLight],
```

```dart
import 'dart:math' as math;

import 'package:flutter/material.dart';

import '../../app/theme/app_theme.dart';

class StarryBackground extends StatelessWidget {
  const StarryBackground({super.key, required this.child});

  final Widget child;

  @override
  Widget build(BuildContext context) {
    return DecoratedBox(
      decoration: const BoxDecoration(
        gradient: LinearGradient(
          begin: Alignment.topLeft,
          end: Alignment.bottomRight,
          colors: [AppTheme.background, AppTheme.backgroundAlt],
        ),
      ),
      child: Stack(
        fit: StackFit.expand,
        children: [
          const CustomPaint(painter: _CosmicBackgroundPainter()),
          const AnimatedStarField(key: Key('global-star-field')),
          child,
        ],
      ),
    );
  }
}

class AnimatedStarField extends StatelessWidget {
  const AnimatedStarField({super.key, this.starCount = 56});

  final int starCount;

  @override
  Widget build(BuildContext context) {
    return IgnorePointer(
      child: LayoutBuilder(
        builder: (context, constraints) {
          return Stack(
            children: List.generate(starCount, (index) {
              final width = math.max(constraints.maxWidth, 1);
              final height = math.max(constraints.maxHeight, 1);
              final left = ((index * 37) % 100) / 100 * width;
              final top = ((index * 53) % 100) / 100 * height;
              final size = 1.2 + (index % 3) * 0.7;

              return Positioned(
                left: left,
                top: top,
                child: _TwinklingStar(index: index, size: size),
              );
            }),
          );
        },
      ),
    );
  }
}

class _TwinklingStar extends StatefulWidget {
  const _TwinklingStar({required this.index, required this.size});

  final int index;
  final double size;

  @override
  State<_TwinklingStar> createState() => _TwinklingStarState();
}

class _TwinklingStarState extends State<_TwinklingStar>
    with SingleTickerProviderStateMixin {
  late final AnimationController _controller;
  late final Animation<double> _opacity;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: Duration(milliseconds: 1200 + widget.index % 8 * 220),
      value: (widget.index % 10) / 10,
    )..repeat(reverse: true);
    _opacity = Tween<double>(begin: 0.16, end: 0.9).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeInOut),
    );
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return FadeTransition(
      opacity: _opacity,
      child: Container(
        width: widget.size,
        height: widget.size,
        decoration: BoxDecoration(
          color: AppTheme.textPrimary,
          borderRadius: BorderRadius.circular(widget.size),
          boxShadow: [
            BoxShadow(
              color: AppTheme.primaryLight.withOpacity(0.45),
              blurRadius: widget.size * 4,
            ),
          ],
        ),
      ),
    );
  }
}

class _CosmicBackgroundPainter extends CustomPainter {
  const _CosmicBackgroundPainter();

  @override
  void paint(Canvas canvas, Size size) {
    _drawGlow(
      canvas,
      size,
      center: Offset(size.width * 0.2, size.height * 0.08),
      radius: size.shortestSide * 0.45,
      color: AppTheme.primary.withOpacity(0.28),
    );
    _drawGlow(
      canvas,
      size,
      center: Offset(size.width * 0.78, size.height * 0.72),
      radius: size.shortestSide * 0.4,
      color: AppTheme.primaryLight.withOpacity(0.1),
    );

    final linePaint = Paint()
      ..color = AppTheme.border.withOpacity(0.2)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1;

    for (var i = 0; i < 7; i++) {
      final y = size.height * (0.16 + i * 0.13);
      final path = Path()
        ..moveTo(-40, y)
        ..cubicTo(
          size.width * 0.25,
          y - 70,
          size.width * 0.62,
          y + 78,
          size.width + 40,
          y - 30,
        );
      canvas.drawPath(path, linePaint);
    }
  }

  void _drawGlow(
    Canvas canvas,
    Size size, {
    required Offset center,
    required double radius,
    required Color color,
  }) {
    final rect = Rect.fromCircle(center: center, radius: radius);
    final paint = Paint()
      ..shader = RadialGradient(
        colors: [color, Colors.transparent],
      ).createShader(rect);
    canvas.drawCircle(center, radius, paint);
  }

  @override
  bool shouldRepaint(covariant _CosmicBackgroundPainter oldDelegate) => false;
```
